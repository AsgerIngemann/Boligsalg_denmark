Boligsalg i Danmark siden 1992-01-01.

Denmarks kort

Dette kort viser boligsalg i Danmark fra den 1. januar 1992 og frem. De røde prikker repræsenterer individuelle ejendomssalg, hvor hver prik angiver en bolig, der er blevet solgt.

Varmekortet viser tætheden af boligsalg i Danmark over tid. De farvede områder viser, hvor mange salg der har fundet sted i forskellige dele af landet. Blå og lilla nuancer indikerer færre salg, mens gule grønne, røde områder markerer høj koncentration af bolighandler. Kortet giver et visuelt overblik over, hvor bolighandlen har været mest aktiv over tid.

import { aq, op } from '@uwdata/arquero'

dt = aq.loadCSV('boligsiden_count.csv');
data = FileAttachment("boligsiden_count.csv").csv({typed: true})
raw_data = aq.loadCSV("houses_sold_all.csv");
postnummer_data = aq.loadCSV("postnummer_data.csv");

data_all = raw_data
    .derive({datenew: d => op.format_utcdate(d.date)})
    .derive({year: d => op.year(d.datenew)})

cityNames_ar = dt
  .rollup({ a: op.array_agg_distinct('cityName') })
dates = data_all
    .orderby('year')
    .rollup({a : op.array_agg_distinct('year')})

year = {
    let i = 0;
    while (true) {
        yield Promises.delay(1000, dates['_data']['a'][0][i++]);
        if (i == dates['_data']['a'][0].length) {
            i = 0;
        }
    }
}

filtered = data_all
    .params({
        y: year
    })
    .filter((d, $) => d.year == $.y)
denmark = FileAttachment("kommuner.geojson").json()

Plot.plot({
    width: 1000,
    height: 1000,
    x: { domain: [7.5,13.5]},    
    marks: [
        Plot.frame({fill: "white", stroke: "white", strokeWidth: 1 }),
        Plot.geo(denmark, {
            stroke: "gray",
            strokeWidth: 1,
            fill: "white"
        }),
        Plot.dot(filtered, {x: "coordinates_lon", y: "coordinates_lat", stroke: "red", fill: "red", fillOpacity: 0.2, r: 2}),
        Plot.density(filtered, {x: 'coordinates_lon', y: 'coordinates_lat', bandwith: 5,
            fill: 'density',
            fillOpacity: 0.2
        }),
        Plot.axisX({ticks: [], label: []}),
        Plot.axisY({ticks: [], label: []}),
        Plot.text([[12.8, 57.7]], {text: [String(year)], textAnchor: "start", fontSize: 24})
    ]
})
viewof selectedOptions = Inputs.checkbox(cityNames_ar['_data']['a'][0], {label: "City name"});
test = dt
    .params({
        names: selectedOptions
    })
    .filter((d,$) => op.includes($.names, d.cityName))
    .orderby(aq.desc('year'))

Plot.plot({
    color: {legend: true},
    marks: [
        Plot.lineY(test, {x: 'year', y: 'n', stroke: "cityName"})
    ]
})

        y: year
    })
